# GUI + plotting
PySide6>=6.5
pyqtgraph>=0.13
qtawesome>=1.3        # vector icons (Material Design Icons) for toolbars/buttons
superqt>=0.6         # nicer widgets: toggle switches, labeled/range sliders
PyOpenGL>=3.1        # 3D stage/setup view (pyqtgraph.opengl GLViewWidget)
PySide6-QtAds>=4.1   # dockable/floating panels (multi-monitor cockpit layout)

# Numerics + data
# Upper-bounded to <2 on purpose: the vendored FLIR PySpin 3.2 wheel is built
# against the numpy 1.x C-ABI and fails to import under numpy 2.x with
# "_ARRAY_API not found / numpy.core.multiarray failed to import". numpy 1.26.x
# is the last 1.x release and is compatible with scipy/pyqtgraph/PySide6/h5py here.
numpy>=1.24,<2
scipy>=1.11
h5py>=3.8
PyYAML>=6.0
Pillow>=10.0

# Instrument I/O
pyvisa>=1.13          # VISA oscilloscope / waveform generator (USB + LAN)
pyserial>=3.5         # GRBL/Marlin motor stage, e4control 'usb' transport
zeroconf>=0.130       # optional: mDNS/LXI auto-discovery of LAN instruments
influxdb-client>=1.36 # optional slow-control sink

# e4control bias-supply transports (vendored package in ./vendor/e4control)
#   TCP/UDP ('lan', 'serial', 'lan_udp') use a built-in socket shim — no dep.
#   'prologix' uses a built-in socket client — no dep.
python-vxi11>=0.9     # only needed for 'gpib' / 'gpibSerial' connection types
